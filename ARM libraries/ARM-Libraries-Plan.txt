POTENTIAL ISSUES AND SOLUTIONS
===============================

ISSUE 1: Gyp doesn't detect ARM64 architecture
SOLUTION: Manually set in environment before config
    export ARCHFLAGS="-arch arm64"
    make config-mac

ISSUE 2: Xcode project still builds x86_64
SOLUTION: Force architecture in xcodebuild command
    xcodebuild ... ARCHS=arm64 ONLY_ACTIVE_ARCH=NO

ISSUE 3: Assembly code doesn't compile
SOLUTION: Check SK_CPU_ARM64 is defined
    Add to defines in libskia.gyp if needed:
    'defines': ['SK_CPU_ARM64']

ISSUE 4: NEON intrinsics not found
SOLUTION: Ensure arm_neon.h is available
    Should be in SDK: thirdparty/sdk/MacOSX12.1.sdk/usr/include/arm_neon.h

ISSUE 5: Linking errors with dependencies
SOLUTION: Build dependencies first (Phase 4)
    Ensure all .a files are ARM64

ISSUE 6: Runtime crashes on ARM Mac
SOLUTION: Check for x86-specific assumptions
    - Verify no SSE intrinsics leaked in
    - Check alignment requirements
    - Validate endianness handling


BUILD SCRIPT TEMPLATE
=====================

Create: ARM-Libraries/build-libskia-arm64.sh

#!/bin/bash
set -e

echo "Building libskia for ARM64 macOS"

# Configuration
LIVECODE_ROOT="/Users/admin/Cascade/Livecode-Community-9.6.3/livecode"
BUILD_DIR="$LIVECODE_ROOT/_build/mac/Release"
OUTPUT_DIR="$LIVECODE_ROOT/../ARM-Libraries/lib"

cd "$LIVECODE_ROOT"

# Step 1: Build dependencies
echo "Building libskia dependencies..."
for lib in libgif libexpat libjpeg libpng libz; do
    echo "  Building $lib..."
    xcodebuild -project "build-mac/livecode/thirdparty/$lib/$lib.xcodeproj" \
               -configuration Release \
               -arch arm64 \
               ARCHS=arm64 \
               ONLY_ACTIVE_ARCH=NO
done

# Step 2: Build libskia optimization targets
echo "Building libskia optimization targets..."
xcodebuild -project "build-mac/livecode/thirdparty/libskia/libskia.xcodeproj" \
           -configuration Release \
           -target libskia_opt_none \
           -arch arm64 \
           ARCHS=arm64 \
           ONLY_ACTIVE_ARCH=NO

xcodebuild -project "build-mac/livecode/thirdparty/libskia/libskia.xcodeproj" \
           -configuration Release \
           -target libskia_opt_arm \
           -arch arm64 \
           ARCHS=arm64 \
           ONLY_ACTIVE_ARCH=NO

# Step 3: Build main libskia
echo "Building libskia..."
xcodebuild -project "build-mac/livecode/thirdparty/libskia/libskia.xcodeproj" \
           -configuration Release \
           -target libskia \
           -arch arm64 \
           ARCHS=arm64 \
           ONLY_ACTIVE_ARCH=NO

# Step 4: Verify and copy
echo "Verifying ARM64 architecture..."
file "$BUILD_DIR/libskia.a"
file "$BUILD_DIR/libskia_opt_arm.a"

echo "Copying to ARM-Libraries..."
mkdir -p "$OUTPUT_DIR"
cp "$BUILD_DIR"/libskia*.a "$OUTPUT_DIR/"

echo "libskia ARM64 build complete!"
echo "Libraries in: $OUTPUT_DIR"
